/*
    this scripts loops over all pairs <g,q> and checks how many have finished
*/
    //AttachSpec("~/CHIMP/CHIMP.spec");
    //AttachSpec("~/AlgEt/spec");
    //AttachSpec("~/AbVarFq_LMFDBLabels/spec");

    SetColumns(0);
    fld_comp:="/data/stmar/287_abvarfq_lmfdb_recomputation/";
    fld_out_pol:=fld_comp * "output_pols/av_fq_isog/";

    printf "Loading input and already-done...\n";
    time all:=Split(Read("~/AbVarFq_LMFDBLabels/computation/weil_poly_sqfree_ord_cs_no_3_25.txt")) cat Split(Read("~/AbVarFq_LMFDBLabels/computation/weil_poly_sqfree_ord_only_3_25.txt"));
    princ_pols_joblog:=Read(fld_comp * "joblog_pols_all_but_3_25");
    time fld_out_pol_ls:=[Join(Split(c,".")[1..2],".") : c in Split(Pipe("ls " * fld_out_pol,"r"))];
    printf "...done!\n";
   
    printf "Extracting <g,q>'s and sorting by number of isogeny classes...\n";
    arr:=AssociativeArray();
    tot:=#all; perc:=0;
    for ic->cc in all do
        if Truncate(ic/tot*100) gt perc then perc+:=1; printf "%o%% ",perc; end if;
        c:=Split(cc,"[ ,]");
        g:=(#c-1) div 2;
        q:=Round((StringToInteger(c[1]))^(1/g));
        ind:=Sprintf("%o.%o",g,q);
        if ind notin Keys(arr) then
            arr[ind]:=1;
        else
            arr[ind]+:=1;
        end if;
    end for;
    printf "...done!\n";

    keys:=Setseq(Keys(arr));
    sizes:=[arr[k]:k in keys];
    ParallelSort(~sizes,~keys);

    done_all_pol:=0;
    for k in keys do
        done_k_pol:=#[c : c in fld_out_pol_ls | c eq k];
        done_all_pol +:=done_k_pol;
        all_k:=arr[k];
        perc_k_pol:=Truncate(done_k_pol*100/all_k);
        printf "<g,q>=%o\ttot =%6o\tdone: pol =%6o\t%o%%\n",k,all_k,done_k_pol,perc_k_pol;
    end for;

    /* for parsing the joblog. not usable anymore after restarting the comp
    princ_pols_joblog:=Split(princ_pols_joblog);
    princ_pols_joblog:=princ_pols_joblog[2..#princ_pols_joblog];
    princ_pols_timed_out:=&+[Split(l,"\t ")[7] eq "-1" select 1 else 0: l in princ_pols_joblog];
    printf "\n\ntotal = %o | princ pols done =%o  %o%% | princ pols timed out =%o %o%%\n",
                                tot,
                                done_all_pol,
                                Truncate(done_all_pol*100/tot),
                                princ_pols_timed_out,
                                Truncate(princ_pols_timed_out*100/tot);
    */

    quit;
